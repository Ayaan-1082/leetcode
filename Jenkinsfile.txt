pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/Ayaan-1082/leetcode.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t greatest-app:latest .'
            }
        }

        stage('Run Container') {
            steps {
                sh '''
                    docker rm -f greatest-container || true
                    echo "===== Running with numbers 5, 10, 3 ====="
                    docker run --name greatest-container greatest-app:latest 5 10 3
                    echo "===== Run complete ====="
                '''
            }
        }

        stage('Display Docker Info') {
            steps {
                sh 'docker images'
                sh 'docker ps -a'
            }
        }
    }
}
